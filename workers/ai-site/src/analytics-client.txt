(function () {
  var $ = function (i) { return document.getElementById(i); };
  var range = new URLSearchParams(location.search).get('r');
  if (['today', '7d', '30d', '90d'].indexOf(range) < 0) range = '7d';
  var metric = 'v', D = null, busy = false;
  var NS = 'http://www.w3.org/2000/svg';
  function num(x) { return (x || 0).toLocaleString('en-US'); }
  function el(tag, cls, text) { var e = document.createElement(tag); if (cls) e.className = cls; if (text != null) e.textContent = text; return e; }
  function sv(tag, a) { var e = document.createElementNS(NS, tag); for (var k in a) e.setAttribute(k, a[k]); return e; }
  function clear(n) { while (n.firstChild) n.removeChild(n.firstChild); return n; }
  var regions; try { regions = new Intl.DisplayNames(['en'], { type: 'region' }); } catch (e) {}
  function flag(cc) { return /^[A-Z]{2}$/.test(cc) && cc !== 'XX' ? String.fromCodePoint(127397 + cc.charCodeAt(0), 127397 + cc.charCodeAt(1)) + ' ' : ''; }
  function country(cc) { try { return cc && cc !== 'XX' && regions ? regions.of(cc) : 'Unknown'; } catch (e) { return cc || 'Unknown'; } }
  var CH = { direct: 'Direct', search: 'Organic search', ai: 'AI assistants', social: 'Social', referral: 'Other websites', email: 'Email', paid: 'Paid ads', internal: 'Our own site' };
  var DEV = { desktop: 'Desktop', mobile: 'Mobile', tablet: 'Tablet' };
  function ago(ts) { var s = Math.max(0, Math.round((Date.now() - ts) / 1000)); if (s < 60) return s + 's ago'; var m = Math.round(s / 60); return m < 60 ? m + ' min ago' : Math.round(m / 60) + ' h ago'; }
  var TZ = 'America/Chicago';
  function fmtT(t, hourly, long) {
    var d = new Date(t);
    if (hourly) return d.toLocaleTimeString('en-US', { timeZone: TZ, hour: 'numeric' });
    return d.toLocaleDateString('en-US', { timeZone: TZ, month: 'short', day: 'numeric', weekday: long ? 'short' : undefined });
  }
  function nice(max) { var p = Math.pow(10, Math.floor(Math.log(Math.max(max, 1)) / Math.LN10)), m = max / p; return (m <= 1 ? 1 : m <= 2 ? 2 : m <= 5 ? 5 : 10) * p; }

  // A hover readout shared by the charts: values lead, labels follow. Text only (textContent), never HTML.
  function tooltip(host) {
    var t = el('div', 'tip'); t.style.display = 'none'; host.appendChild(t);
    return {
      show: function (x, y, title, rows) {
        clear(t); t.appendChild(el('div', 'k', title));
        rows.forEach(function (r) { var d = el('div'); var k = el('span', r.dash ? 'key p' : 'key'); d.appendChild(k); var b = el('b', null, r.value); d.appendChild(b); d.appendChild(document.createTextNode(' ' + r.label)); t.appendChild(d); });
        t.style.display = 'block';
        var w = t.offsetWidth, hw = host.clientWidth;
        t.style.left = Math.max(0, Math.min(hw - w, x + 14)) + 'px'; t.style.top = Math.max(0, y - 10) + 'px';
      },
      hide: function () { t.style.display = 'none'; }
    };
  }

  // Bars: users per minute, last 30 minutes (the "realtime" strip).
  function bars(host, mins) {
    clear(host); var W = host.clientWidth || 400, H = 150, l = 4, r = 4, t = 8, b = 20, max = nice(Math.max.apply(null, mins.map(function (m) { return m.u; }).concat([1])));
    var s = sv('svg', { viewBox: '0 0 ' + W + ' ' + H, width: '100%', height: H, role: 'img', 'aria-label': 'Users per minute over the last 30 minutes' });
    var bw = (W - l - r) / mins.length, tp = tooltip(host);
    mins.forEach(function (m, i) {
      var h = m.u ? Math.max(3, (H - t - b) * m.u / max) : 0, x = l + i * bw;
      if (h) s.appendChild(sv('rect', { x: x + 1, y: H - b - h, width: Math.max(2, bw - 2), height: h, rx: 2, fill: 'var(--s1)' }));
      var hit = sv('rect', { x: x, y: 0, width: bw, height: H - b, fill: 'transparent', tabindex: 0 });
      var say = function () { tp.show(x + bw / 2, 20, new Date(m.t).toLocaleTimeString('en-US', { timeZone: TZ, hour: 'numeric', minute: '2-digit' }), [{ label: m.u === 1 ? 'user' : 'users', value: num(m.u) }]); };
      hit.addEventListener('pointermove', say); hit.addEventListener('focus', say); hit.addEventListener('pointerleave', tp.hide); hit.addEventListener('blur', tp.hide);
      s.appendChild(hit);
    });
    s.appendChild(sv('line', { x1: l, x2: W - r, y1: H - b + .5, y2: H - b + .5, stroke: 'var(--grid)' }));
    [[0, '30 min ago'], [14, '15'], [29, 'now']].forEach(function (p) { var tx = sv('text', { x: l + p[0] * bw + (p[0] === 29 ? bw : p[0] ? bw / 2 : 0), y: H - 5, 'text-anchor': p[0] === 29 ? 'end' : p[0] ? 'middle' : 'start' }); tx.textContent = p[1]; s.appendChild(tx); });
    host.insertBefore(s, host.firstChild);
  }

  // Line chart: this period (solid) against the previous period (dashed), crosshair + tooltip, arrow keys work too.
  function line(host, cur, prev, key, hourly, name) {
    clear(host); var W = host.clientWidth || 700, H = 270, l = 44, r = 14, t = 12, b = 28, n = prev.length, m = cur.length;   // m < n: today, hours that have not happened yet are not drawn
    var span = hourly ? 36e5 : 864e5, prevT = function (i) { return D.from + i * span; };
    var vals = cur.map(function (p) { return p[key]; }), pv = prev.map(function (p) { return p[key]; });
    var max = nice(Math.max.apply(null, vals.concat(pv, [1]))), X = function (i) { return n === 1 ? (l + W - r) / 2 : l + (W - l - r) * i / (n - 1); }, Y = function (v) { return t + (H - t - b) * (1 - v / max); };
    var s = sv('svg', { viewBox: '0 0 ' + W + ' ' + H, width: '100%', height: H, role: 'img', tabindex: 0, 'aria-label': name + ' over time. Use the left and right arrow keys to read each point.' });
    for (var g = 0; g <= 4; g++) { var gy = Y(max * g / 4); s.appendChild(sv('line', { x1: l, x2: W - r, y1: gy, y2: gy, stroke: 'var(--grid)', 'stroke-width': 1 })); var tx = sv('text', { x: l - 8, y: gy + 4, 'text-anchor': 'end' }); tx.textContent = num(Math.round(max * g / 4)); s.appendChild(tx); }
    var step = Math.max(1, Math.ceil(n / Math.max(2, Math.floor((W - l - r) / 74))), 0);
    var last = Math.floor((n - 1) / step) * step;
    for (var i = 0; i < n; i += step) { var lb = sv('text', { x: X(i), y: H - 8, 'text-anchor': i === 0 ? 'start' : (i === last && i === n - 1) ? 'end' : 'middle' }); lb.textContent = fmtT(prevT(i), hourly); s.appendChild(lb); }
    var path = function (a) { return a.map(function (v, i) { return (i ? 'L' : 'M') + X(i).toFixed(1) + ' ' + Y(v).toFixed(1); }).join(' '); };
    s.appendChild(sv('path', { d: path(vals) + ' L' + X(m - 1).toFixed(1) + ' ' + Y(0) + ' L' + X(0).toFixed(1) + ' ' + Y(0) + ' Z', fill: 'var(--s1)', opacity: 0.1 }));
    s.appendChild(sv('path', { d: path(pv), fill: 'none', stroke: 'var(--muted)', 'stroke-width': 2, 'stroke-dasharray': '5 4' }));
    s.appendChild(sv('path', { d: path(vals), fill: 'none', stroke: 'var(--s1)', 'stroke-width': 2, 'stroke-linejoin': 'round', 'stroke-linecap': 'round' }));
    var cross = sv('line', { y1: t, y2: H - b, stroke: 'var(--muted)', 'stroke-width': 1, visibility: 'hidden' }), dot = sv('circle', { r: 4.5, fill: 'var(--s1)', stroke: '#fff', 'stroke-width': 2, visibility: 'hidden' });
    s.appendChild(cross); s.appendChild(dot);
    var tp = tooltip(host), cur_i = -1;
    function at(i) {
      i = Math.max(0, Math.min(m - 1, i)); cur_i = i;
      cross.setAttribute('x1', X(i)); cross.setAttribute('x2', X(i)); cross.setAttribute('visibility', 'visible');
      dot.setAttribute('cx', X(i)); dot.setAttribute('cy', Y(vals[i])); dot.setAttribute('visibility', 'visible');
      tp.show(X(i), Y(vals[i]), fmtT(prevT(i), hourly, true), [{ label: name.toLowerCase(), value: num(vals[i]) }, { label: 'previous period', value: num(pv[i]), dash: true }]);
    }
    var hit = sv('rect', { x: l, y: t, width: W - l - r, height: H - t - b, fill: 'transparent' });
    hit.addEventListener('pointermove', function (e) { var bx = s.getBoundingClientRect(); var px = (e.clientX - bx.left) * W / bx.width; at(Math.round((px - l) / ((W - l - r) / Math.max(1, n - 1)))); });
    hit.addEventListener('pointerleave', function () { cross.setAttribute('visibility', 'hidden'); dot.setAttribute('visibility', 'hidden'); tp.hide(); });
    s.addEventListener('keydown', function (e) { if (e.key === 'ArrowRight') { at(cur_i + 1); e.preventDefault(); } else if (e.key === 'ArrowLeft') { at(cur_i < 0 ? m - 1 : cur_i - 1); e.preventDefault(); } else if (e.key === 'Escape') hit.dispatchEvent(new Event('pointerleave')); });
    s.appendChild(hit); host.insertBefore(s, host.firstChild);
  }

  function list(id, rows, label, sub) {
    var h = clear($(id));
    if (!rows || !rows.length) { h.appendChild(el('div', 'empty', 'Nothing yet.')); return; }
    var max = Math.max.apply(null, rows.map(function (r) { return r.pv; }).concat([1]));
    rows.forEach(function (r) {
      var d = el('div', 'li'); var k = el('span', 'k', label(r)); k.title = label(r); d.appendChild(k); d.appendChild(el('span', 'n', num(r.pv) + (sub ? ' · ' + num(r.v) : '')));
      var b = el('div', 'b'); var i = el('i'); i.style.width = Math.max(2, Math.round(100 * r.pv / max)) + '%'; b.appendChild(i); d.appendChild(b); h.appendChild(d);
    });
  }
  function devch(rows) {
    var h = clear($('devch'));
    if (!rows || !rows.length) { h.appendChild(el('div', 'empty', 'Nothing yet.')); return; }
    var pct = function (a, n) { return n ? Math.round(100 * a / n) + '%' : '0%'; };
    rows.forEach(function (r) {
      var d = el('div', 'li'); var k = el('span', 'k', CH[r.k] || r.k); d.appendChild(k);
      d.appendChild(el('span', 'n', 'Desktop ' + pct(r.desktop, r.v) + ' · Mobile ' + pct(r.mobile, r.v) + ' · Tablet ' + pct(r.tablet, r.v) + '  (' + num(r.v) + ' visitors)'));
      var b = el('div', 'b'); var i = el('i'); i.style.width = Math.max(2, Math.round(100 * r.mobile / Math.max(r.v, 1))) + '%'; b.appendChild(i); d.appendChild(b); h.appendChild(d);
    });
  }
  function delta(cur, prev) { if (!prev) return cur ? 'new' : '–'; var p = Math.round(100 * (cur - prev) / prev); return (p > 0 ? '▲ ' : p < 0 ? '▼ ' : '') + Math.abs(p) + '% vs previous'; }
  function tiles() {
    var h = clear($('tiles')), T = D.totals, P = D.prevTotals;
    var ppv = T.v ? T.pv / T.v : 0, pp = P.v ? P.pv / P.v : 0;
    [['v', 'Visitors', T.v, P.v], ['pv', 'Page views', T.pv, P.pv], [null, 'Pages per visitor', Math.round(ppv * 10) / 10, Math.round(pp * 10) / 10]].forEach(function (x) {
      var b = el(x[0] ? 'button' : 'div', 'tile' + (x[0] ? '' : ' static')); if (x[0]) { b.type = 'button'; b.setAttribute('aria-pressed', String(metric === x[0])); b.onclick = function () { metric = x[0]; render(); }; }
      b.appendChild(el('div', 'sub', x[1])); b.appendChild(el('div', 'v', x[2].toLocaleString('en-US'))); b.appendChild(el('div', 'delta', delta(x[2], x[3]))); h.appendChild(b);
    });
  }
  function table(host, rows, hourly) {
    clear(host); var t = el('table'), hd = el('tr'); ['When', 'Visitors', 'Page views', 'Previous visitors', 'Previous page views'].forEach(function (c) { hd.appendChild(el('th', null, c)); }); t.appendChild(hd);
    rows.forEach(function (p, i) { var tr = el('tr'); [fmtT(p.t, hourly, true), num(p.v), num(p.pv), num(D.prev[i].v), num(D.prev[i].pv)].forEach(function (c) { tr.appendChild(el('td', null, c)); }); t.appendChild(tr); }); host.appendChild(t);
  }
  function render() {
    var hourly = D.range === 'today', name = metric === 'v' ? 'Visitors' : 'Page views';
    $('since').textContent = D.since ? 'Counting since ' + new Date(D.since).toLocaleDateString('en-US', { timeZone: TZ, month: 'long', day: 'numeric', year: 'numeric' }) + '. Earlier visits were not recorded.' : 'No visits recorded yet. Counting started when this page went live.';
    document.querySelectorAll('.filters button[data-r]').forEach(function (b) { b.setAttribute('aria-pressed', String(b.dataset.r === range)); });
    var rt = D.rt; $('rtnum').textContent = num(rt.users30); $('rt5').textContent = num(rt.users5); bars($('rtbars'), rt.minutes);
    list('rtpages', rt.pages, function (r) { return r.k; }); list('rtch', rt.channels, function (r) { return CH[r.k] || r.k; });
    var f = clear($('rtlatest')); if (!rt.latest.length) f.appendChild(el('div', 'empty', 'Waiting for the first visit…'));
    rt.latest.forEach(function (v) { var d = el('div'); d.appendChild(el('span', null, flag(v.country) + v.path)); d.appendChild(el('span', 'sub', ago(v.ts))); f.appendChild(d); });
    tiles();
    var lg = clear($('legend')); var a = el('span', 'key'); lg.appendChild(a); lg.appendChild(document.createTextNode(name + ', this period   ')); var b = el('span', 'key p'); lg.appendChild(b); lg.appendChild(document.createTextNode('previous period'));
    line($('main'), hourly ? D.series.filter(function (p) { return p.t <= D.now; }) : D.series, D.prev, metric, hourly, name); table($('maint'), D.series, hourly);
    list('pages', D.pages, function (r) { return r.k; }, true);
    list('channels', D.channels, function (r) { return CH[r.k] || r.k; }, true); list('referrers', D.referrers, function (r) { return r.k; }, true);
    list('countries', D.countries, function (r) { return flag(r.k) + country(r.k); }, true); list('devices', D.devices, function (r) { return DEV[r.k] || r.k; }, true); list('browsers', D.browsers, function (r) { return r.k; }, true); devch(D.deviceByChannel);
    $('upd').textContent = 'live · updated ' + new Date().toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', second: '2-digit' });
  }
  function load(fresh) {
    if (busy) return; busy = true; document.querySelector('.an').classList.add('stale');
    fetch('/blog-admin/analytics/data?range=' + range + (fresh === true ? '&fresh=1' : ''), { credentials: 'same-origin' }).then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
      .then(function (d) { D = d; render(); if (fresh === true) $('upd').textContent = d.throttled ? 'just refreshed, wait a few seconds' : 'refreshed ' + new Date().toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', second: '2-digit' }); })
      .catch(function (e) { $('upd').textContent = 'could not refresh (' + e.message + '). Press Refresh now.'; })
      .then(function () { busy = false; document.querySelector('.an').classList.remove('stale'); });
  }
  // ---- Connected services: one card per provider, small bars per metric ----
  var SV = null;
  function money(v) { return '$' + (v || 0).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); }
  function fmtV(v, unit) { return v == null ? '–' : unit === 'usd' ? money(v) : num(Math.round(v * 100) / 100); }
  function miniBars(host, series, unit, label) {
    var W = 150, H = 38, mx = Math.max.apply(null, series.map(function (x) { return x.v; }).concat([0.0001])), bw = W / series.length;
    var s = sv('svg', { viewBox: '0 0 ' + W + ' ' + H, width: '100%', height: H, role: 'img', 'aria-label': label + ' per day' }), tp = tooltip(host);
    series.forEach(function (x, i) {
      var h = x.v ? Math.max(2, (H - 2) * x.v / mx) : 0;
      if (h) s.appendChild(sv('rect', { x: i * bw + 0.5, y: H - h, width: Math.max(1, bw - 1), height: h, rx: 1, fill: 'var(--s1)' }));
      var hit = sv('rect', { x: i * bw, y: 0, width: bw, height: H, fill: 'transparent' });
      var say = function () { tp.show((i + .5) * (host.clientWidth / series.length), 10, new Date(x.day + 'T12:00:00Z').toLocaleDateString('en-US', { timeZone: 'UTC', month: 'short', day: 'numeric' }), [{ label: label.toLowerCase(), value: fmtV(x.v, unit) }]); };
      hit.addEventListener('pointermove', say); hit.addEventListener('pointerleave', tp.hide); s.appendChild(hit);
    });
    s.setAttribute('preserveAspectRatio', 'none'); host.insertBefore(s, host.firstChild);
  }
  function services() {
    var h = clear($('svc')); if (!SV) return;
    var groups = []; SV.providers.forEach(function (p) { if (groups.indexOf(p.group) < 0) groups.push(p.group); });
    groups.forEach(function (g) {
      h.appendChild(el('h3', 'gh', g)); var grid = el('div', 'svc');
      SV.providers.filter(function (p) { return p.group === g; }).forEach(function (p) {
        var c = el('div', 'card'), top = el('div'); top.style.cssText = 'display:flex;justify-content:space-between;gap:8px;align-items:center';
        top.appendChild(el('b', null, p.name));
        var ok = p.state === 'ok', chip = el('span', 'chip ' + (ok ? 'ok' : p.state === 'error' ? 'bad' : ''), ok ? 'Connected' : p.state === 'error' ? 'Error' : p.state === 'needs_key' ? 'Needs a key' : p.state === 'not_connected' ? 'Not connected' : 'Waiting');
        top.appendChild(chip); c.appendChild(top);
        c.appendChild(el('div', 'sub', ok ? 'Updated ' + (p.at ? ago(p.at * 1000) : 'just now') : (p.message || 'Waiting for the first collection (runs every 15 minutes).')));
        if (ok && p.metrics.length && !p.metrics.some(function (m) { return m.has_data; })) c.appendChild(el('div', 'sub', 'No activity recorded yet. Numbers appear here as soon as this provider is used.'));
        else if (p.metrics.length && ok) {
          var mt = el('div', 'mt');
          p.metrics.forEach(function (m) {
            var b = el('div'); b.appendChild(el('div', 'sub', m.label + (m.how === 'latest' ? ' (now)' : '')));
            b.appendChild(el('div', 'v', m.has_data ? fmtV(m.value, m.unit) : '–'));
            var ch = el('div', 'chart'); b.appendChild(ch); mt.appendChild(b); if (m.has_data) miniBars(ch, m.series, m.unit, m.label);
          });
          c.appendChild(mt);
        }
        grid.appendChild(c);
      });
      h.appendChild(grid);
    });
  }
  function loadServices() {
    var d = { today: 7, '7d': 7, '30d': 30, '90d': 90 }[range] || 30;
    fetch('/blog-admin/analytics/providers?days=' + d, { credentials: 'same-origin' }).then(function (r) { return r.json(); })
      .then(function (x) { if (x.error) throw new Error(x.error); SV = x; services(); }).catch(function (e) { $('svcnote').textContent = 'Could not load the connected services: ' + e.message; });
  }
  $('collectnow').onclick = function () {
    var b = $('collectnow'); b.disabled = true; $('collectmsg').textContent = 'Collecting from every provider…';
    fetch('/blog-admin/analytics/collect', { method: 'POST', credentials: 'same-origin', headers: { 'Content-Type': 'application/json', 'X-CSRF': CSRF }, body: '{}' })
      .then(function (r) { return r.json().catch(function () { return {}; }); }).then(function (x) { $('collectmsg').textContent = x.error || (x.ok === false ? 'Could not collect.' : 'Done.'); loadServices(); })
      .catch(function () { $('collectmsg').textContent = 'Could not collect right now.'; }).then(function () { b.disabled = false; });
  };
  setInterval(function () { if (!document.hidden) loadServices(); }, 60000);

  document.querySelectorAll('.filters button[data-r]').forEach(function (b) { b.onclick = function () { range = b.dataset.r; history.replaceState(null, '', '?r=' + range); load(); loadServices(); }; });
  $('refresh').onclick = function () { var b = $('refresh'); b.disabled = true; b.textContent = 'Refreshing…'; $('upd').textContent = 'refreshing…'; busy = false; load(true); loadServices(); setTimeout(function () { b.disabled = false; b.textContent = '⟳ Refresh now'; }, 3000); };
  window.addEventListener('online', function () { load(); loadServices(); });
  window.addEventListener('resize', function () { if (D) render(); });
  setInterval(function () { if (!document.hidden) load(); }, 15000);
  document.addEventListener('visibilitychange', function () { if (!document.hidden) load(); });
  load(); loadServices();
})();
